cmake_minimum_required(VERSION 3.20)
project(ajin_lidar LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(core_test tests/core_test.cpp)
target_include_directories(core_test PRIVATE include)
add_test(NAME core_test COMMAND core_test)
option(AJIN_CORE_ONLY "Build hardware-independent ring/conversion tests only" OFF)
if(AJIN_CORE_ONLY)
 return()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "Production S2E driver targets Linux")
endif()
include(FetchContent)
FetchContent_Declare(rplidar GIT_REPOSITORY https://github.com/Slamtec/rplidar_sdk.git GIT_TAG 99478e5fb90de3b4a6db0080acacd373f8b36869)
FetchContent_MakeAvailable(rplidar)
set(SDK ${rplidar_SOURCE_DIR}/sdk)
file(GLOB UNPACKER CONFIGURE_DEPENDS ${SDK}/src/dataunpacker/*.cpp ${SDK}/src/dataunpacker/unpacker/*.cpp)
add_library(slamtec STATIC ${SDK}/src/sl_lidar_driver.cpp ${SDK}/src/hal/thread.cpp ${SDK}/src/sl_crc.cpp ${SDK}/src/sl_serial_channel.cpp ${SDK}/src/sl_lidarprotocol_codec.cpp ${SDK}/src/sl_async_transceiver.cpp ${SDK}/src/sl_tcp_channel.cpp ${SDK}/src/sl_udp_channel.cpp ${SDK}/src/arch/linux/net_serial.cpp ${SDK}/src/arch/linux/net_socket.cpp ${SDK}/src/arch/linux/timer.cpp ${UNPACKER})
target_include_directories(slamtec PUBLIC ${SDK}/include PRIVATE ${SDK}/src ${SDK}/src/dataunpacker ${SDK}/src/dataunpacker/unpacker)
target_link_libraries(slamtec PUBLIC pthread)
find_package(Protobuf REQUIRED)
find_package(gRPC CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
set(PROTO ${CMAKE_CURRENT_SOURCE_DIR}/../../contracts/lidar/v1/lidar.proto)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/lidar.pb.cc ${CMAKE_CURRENT_BINARY_DIR}/lidar.pb.h ${CMAKE_CURRENT_BINARY_DIR}/lidar.grpc.pb.cc ${CMAKE_CURRENT_BINARY_DIR}/lidar.grpc.pb.h
 COMMAND protobuf::protoc --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/../../contracts/lidar/v1 --cpp_out=${CMAKE_CURRENT_BINARY_DIR} --grpc_out=${CMAKE_CURRENT_BINARY_DIR} --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> ${PROTO} DEPENDS ${PROTO})
add_executable(lidar-driver src/main.cpp ${CMAKE_CURRENT_BINARY_DIR}/lidar.pb.cc ${CMAKE_CURRENT_BINARY_DIR}/lidar.grpc.pb.cc)
target_include_directories(lidar-driver PRIVATE include ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(lidar-driver PRIVATE slamtec gRPC::grpc++ protobuf::libprotobuf nlohmann_json::nlohmann_json)
install(TARGETS lidar-driver RUNTIME DESTINATION bin)
